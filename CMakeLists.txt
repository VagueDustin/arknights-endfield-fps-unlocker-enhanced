cmake_minimum_required(VERSION 3.24)
project(EndfieldEnhancer VERSION 0.2.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with the x64 MSVC Windows toolchain")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook
  GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
  GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
set(BUILD_SHARED_LIBS OFF)
FetchContent_MakeAvailable(minhook)
add_library(endfield_fps SHARED src/modern/runtime.cpp)
target_link_libraries(endfield_fps PRIVATE minhook user32)
add_library(d3dcompiler_47 SHARED src/modern/loader.cpp src/modern/compiler.def)
foreach(target endfield_fps d3dcompiler_47)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
endforeach()
install(TARGETS endfield_fps d3dcompiler_47 RUNTIME DESTINATION .)
install(FILES tools/manage.py tools/check_compatibility.py DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
install(FILES ${minhook_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME MinHook.txt)
