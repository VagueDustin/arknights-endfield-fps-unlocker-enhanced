cmake_minimum_required(VERSION 3.24)
# One version for the app, the installer, and the runtime log: assets/identity/product.json.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/identity/product.json" product_json)
string(JSON product_version GET "${product_json}" version)
project(FateEngine VERSION ${product_version} LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with the x64 MSVC Windows toolchain")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook
  GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
  GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
set(BUILD_SHARED_LIBS OFF)
FetchContent_MakeAvailable(minhook)
add_library(endfield_fps SHARED src/modern/runtime.cpp src/modern/graphics.cpp)
target_link_libraries(endfield_fps PRIVATE minhook user32 bcrypt)
target_compile_definitions(endfield_fps PRIVATE "FATE_ENGINE_VERSION=\"${PROJECT_VERSION}\"")
add_library(d3dcompiler_47 SHARED src/modern/loader.cpp)
file(STRINGS src/modern/compiler.def compiler_exports REGEX "=")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiler_exports.h" "// Generated from compiler.def\n")
foreach(export IN LISTS compiler_exports)
  string(STRIP "${export}" export)
  string(REPLACE " PRIVATE" "" export "${export}")
  string(REPLACE " @" ",@" export "${export}")
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/compiler_exports.h"
    "#pragma comment(linker, \"/export:${export}\")\n")
endforeach()
target_include_directories(d3dcompiler_47 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
foreach(target endfield_fps d3dcompiler_47)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
endforeach()
install(TARGETS endfield_fps d3dcompiler_47 RUNTIME DESTINATION .)
install(FILES tools/manage.py tools/check_compatibility.py tools/desktop.py tools/desktop_legacy.py tools/desktop_state.py tools/installs.py tools/live_status.py tools/neural.py tools/reshade_setup.py tools/native_neural.py DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
install(DIRECTORY licenses/ DESTINATION licenses)
install(FILES ${minhook_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME MinHook.txt)
include(CTest)
if(BUILD_TESTING)
  add_library(GameAssembly SHARED tests/native/fake_runtime.cpp)
  add_executable(runtime_test tests/native/runtime_test.cpp)
  set_target_properties(runtime_test PROPERTIES OUTPUT_NAME Endfield)
  target_compile_definitions(runtime_test PRIVATE UNICODE _UNICODE)
  add_dependencies(runtime_test GameAssembly endfield_fps d3dcompiler_47)
  add_test(NAME runtime_integration COMMAND runtime_test)
  set_tests_properties(runtime_integration PROPERTIES TIMEOUT 60)
  add_executable(graphics_test tests/native/graphics_test.cpp src/modern/graphics.cpp)
  target_include_directories(graphics_test PRIVATE src/modern)
  target_compile_definitions(graphics_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(graphics_test PRIVATE minhook bcrypt)
  add_dependencies(graphics_test GameAssembly)
  add_test(NAME graphics_controls COMMAND graphics_test)
  add_test(NAME graphics_unknown_build COMMAND graphics_test unknown)
  add_test(NAME graphics_bad_signature COMMAND graphics_test signature)
  add_test(NAME graphics_value_type_rejected COMMAND graphics_test reference)
  set_tests_properties(graphics_controls PROPERTIES TIMEOUT 40)
endif()
